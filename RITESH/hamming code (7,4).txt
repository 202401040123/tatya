#include <iostream>
#include <string>
#include <vector>
using namespace std;

// Hamming (7,4), even parity
void hammingScheme()
{
    string d;
    cout << "Enter 4 data bits : ";
    cin >> d;

    int c[8] = {0};

    c[3] = d[0] - '0';
    c[5] = d[1] - '0';
    c[6] = d[2] - '0';
    c[7] = d[3] - '0';

    c[1] = c[3] ^ c[5] ^ c[7];
    c[2] = c[3] ^ c[6] ^ c[7];
    c[4] = c[5] ^ c[6] ^ c[7];

    cout << "\n--- SENDER ---\n";
    cout << "P1 = " << c[1] << ", P2 = " << c[2]
         << ", P4 = " << c[4] << "\n";

    cout << "Codeword (position 1 to 7) : ";
    for (int i = 1; i <= 7; i++)
        cout << c[i];

    cout << "\n";

    string r;

    cout << "\n--- RECEIVER ---\n";
    cout << "Enter received codeword (position 1 to 7) : ";
    cin >> r;

    for (int i = 1; i <= 7; i++)
        c[i] = r[i - 1] - '0';

    int syndrome = 0;

    for (int i = 1; i <= 7; i++)
        if (c[i])
            syndrome ^= i;

    cout << "Syndrome (S4 S2 S1) = "
         << ((syndrome >> 2) & 1)
         << ((syndrome >> 1) & 1)
         << (syndrome & 1)
         << " -> error position " << syndrome << "\n";

    if (syndrome == 0)
        cout << "Result : no error detected\n";
    else
    {
        c[syndrome] ^= 1;

        cout << "Result : single-bit error at position "
             << syndrome << ", corrected\n";

        cout << "Corrected codeword : ";
        for (int i = 1; i <= 7; i++)
            cout << c[i];

        cout << "\n";
    }

    cout << "Data bits : "
         << c[3] << c[5] << c[6] << c[7] << "\n";
}

int main()
{
    cout << "HAMMING CODE (7,4)\n\n";
    hammingScheme();
    return 0;
}






































Hamming Code ek error detection aur correction technique hai.



Isme 4 data bits aur 3 parity bits use hoti hain.



Total 7 bits ka codeword generate hota hai.



Parity bits positions 1, 2 aur 4 par rakhi jaati hain.



Receiver syndrome calculate karke error ki position find karta hai.



Single-bit error ko identify aur correct kar sakte hain.
