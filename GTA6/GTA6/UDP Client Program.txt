
#include <iostream>
#include <cstring>
#include <sys/socket.h>
#include <arpa/inet.h>
#include <unistd.h>
using namespace std;

int main() {
    int client;
    char buffer[1024] = {0};

    // Create socket
    client = socket(AF_INET, SOCK_DGRAM, 0);

    sockaddr_in server;
    server.sin_family = AF_INET;
    server.sin_port = htons(8080);

    inet_pton(AF_INET, "127.0.0.1", &server.sin_addr);

    socklen_t len = sizeof(server);

    // Send message
    char msg[] = "Hello Server!";
    sendto(client, msg, strlen(msg), 0,
           (struct sockaddr*)&server, len);

    // Receive message
    recvfrom(client, buffer, sizeof(buffer), 0,
             (struct sockaddr*)&server, &len);

    cout << "Server: " << buffer << endl;

    close(client);

    return 0;
}
