
#include <iostream>
#include <string>
using namespace std;

string xorDivision(string data, string divisor) {
    int n = divisor.length();

    for (int i = 0; i <= data.length() - n; i++) {

        if (data[i] == '1') {

            for (int j = 0; j < n; j++) {

                if (data[i + j] == divisor[j])
                    data[i + j] = '0';
                else
                    data[i + j] = '1';
            }
        }
    }

    return data.substr(data.length() - n + 1);
}

int main() {
    string data, divisor, received;

    // Input: Data = 1101, Generator = 1011
    cout << "Enter data bits: ";
    cin >> data;

    cout << "Enter generator bits: ";
    cin >> divisor;

    string temp = data;

    for (int i = 0; i < divisor.length() - 1; i++)
        temp += '0';

    string crc = xorDivision(temp, divisor);

    string transmitted = data + crc;

    cout << "CRC Bits: " << crc << endl;
    cout << "Transmitted Data: " << transmitted << endl;

    // Input: 1101001 (no error)
    cout << "Enter received data: ";
    cin >> received;

    string remainder = xorDivision(received, divisor);

    int error = 0;

    for (int i = 0; i < remainder.length(); i++) {
        if (remainder[i] == '1')
            error = 1;
    }

    if (error == 0)
        cout << "No Error Detected";
    else
        cout << "Error Detected";

    return 0;
}
