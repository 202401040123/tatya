
#include <iostream>
using namespace std;

int main() {
    int n, w, lost;

    // Input: n = 6, window = 3, lost = 2
    cout << "Enter number of frames: ";
    cin >> n;

    cout << "Enter window size: ";
    cin >> w;

    cout << "Enter lost frame (0 for none): ";
    cin >> lost;

    int i = 1;

    while (i <= n) {

        int end = i + w - 1;

        if (end > n)
            end = n;

        cout << "\nSending Frames: ";

        for (int j = i; j <= end; j++)
            cout << j << " ";

        cout << endl;

        if (lost >= i && lost <= end) {

            cout << "Frame " << lost << " Lost!" << endl;

            for (int j = lost; j <= end; j++)
                cout << "Retransmitting Frame " << j << endl;
        }

        else {
            cout << "All Frames Acknowledged";
        }

        i = end + 1;
    }

    cout << "\nTransmission Completed";

    return 0;
}
