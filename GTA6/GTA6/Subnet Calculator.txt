
#include <iostream>
#include <cmath>
using namespace std;

int main() {
    int a, b, c, d, prefix;

    // Input: 192 168 1 10 and prefix 24
    cout << "Enter IP Address: ";
    cin >> a >> b >> c >> d;

    cout << "Enter CIDR Prefix: ";
    cin >> prefix;

    if (a < 0 || a > 255 || b < 0 || b > 255 ||
        c < 0 || c > 255 || d < 0 || d > 255 ||
        prefix < 0 || prefix > 32) {

        cout << "Invalid Input";
        return 0;
    }

    unsigned int ip = (a << 24) | (b << 16) |
                      (c << 8) | d;

    unsigned int mask = 0;

    if (prefix != 0)
        mask = 0xFFFFFFFF << (32 - prefix);

    unsigned int network = ip & mask;
    unsigned int broadcast = network | (~mask);

    cout << "\nSubnet Mask: "
         << ((mask >> 24) & 255) << "."
         << ((mask >> 16) & 255) << "."
         << ((mask >> 8) & 255) << "."
         << (mask & 255);

    cout << "\nNetwork ID: "
         << ((network >> 24) & 255) << "."
         << ((network >> 16) & 255) << "."
         << ((network >> 8) & 255) << "."
         << (network & 255);

    cout << "\nBroadcast Address: "
         << ((broadcast >> 24) & 255) << "."
         << ((broadcast >> 16) & 255) << "."
         << ((broadcast >> 8) & 255) << "."
         << (broadcast & 255);

    cout << "\nTotal IP Addresses: "
         << pow(2, 32 - prefix);

    if (prefix <= 30)
        cout << "\nUsable Hosts: "
             << pow(2, 32 - prefix) - 2;
    else if (prefix == 31)
        cout << "\nUsable Addresses: 2 (point-to-point)";
    else
        cout << "\nUsable Addresses: 1";

    return 0;
}
