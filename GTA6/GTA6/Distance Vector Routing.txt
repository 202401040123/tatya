
#include <iostream>
using namespace std;

int main() {
    int n;
    int cost[10][10], dist[10][10];

    // Input: 4 routers
    // 0 2 999 1
    // 2 0 3 999
    // 999 3 0 4
    // 1 999 4 0

    cout << "Enter number of routers: ";
    cin >> n;

    cout << "Enter cost matrix (999 for infinity):\n";

    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            cin >> cost[i][j];
            dist[i][j] = cost[i][j];
        }
    }

    // Distance Vector calculation
    for (int k = 0; k < n; k++) {

        for (int i = 0; i < n; i++) {

            for (int j = 0; j < n; j++) {

                if (dist[i][k] != 999 && dist[k][j] != 999 &&
                    dist[i][k] + dist[k][j] < dist[i][j]) {

                    dist[i][j] = dist[i][k] + dist[k][j];
                }
            }
        }
    }

    cout << "\nFinal Routing Tables:\n";

    for (int i = 0; i < n; i++) {

        cout << "\nRouter " << i + 1 << endl;

        for (int j = 0; j < n; j++) {
            cout << "Destination " << j + 1
                 << " Cost: " << dist[i][j] << endl;
        }
    }

    return 0;
}
